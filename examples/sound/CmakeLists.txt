# Audio Example

add_executable(sound_example "sound.cpp")
target_link_libraries(sound_example PRIVATE clyde)
target_include_directories(sound_example PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../include")

# Set working directory for debugging
set_target_properties(sound_example PROPERTIES
    VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
)

# Copy asset files to build directory
add_custom_command(TARGET sound_example POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${CMAKE_CURRENT_SOURCE_DIR}/assets"
        "${CMAKE_BINARY_DIR}/assets"
    COMMENT "Copying audio assets..."
)
